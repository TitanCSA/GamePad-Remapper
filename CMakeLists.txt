cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(GamePadRemapper VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
  message(FATAL_ERROR "Use the Visual Studio 2022 x64 C++ toolchain on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT "/PDBALTPATH:%_PDB%")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/obj/lib")
endforeach()
add_library(InterceptorSDK STATIC Interceptor/Interceptor.cpp)
target_include_directories(InterceptorSDK PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Interceptor")
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_library(GamePadRemapper.Native SHARED Native/Remapper.cpp)
target_compile_definitions(GamePadRemapper.Native PRIVATE GR_BUILD_DLL)
target_link_libraries(GamePadRemapper.Native PRIVATE InterceptorSDK)
add_executable(RemapperNativeTests Native/PureTests.cpp)
target_include_directories(RemapperNativeTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/Interceptor")
add_executable(RemapperLiveTests Native/LiveTests.cpp)
target_link_libraries(RemapperLiveTests PRIVATE GamePadRemapper.Native)
target_link_options(RemapperLiveTests PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
enable_testing()
add_test(NAME NativePure COMMAND RemapperNativeTests)
