cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(GamePadRemapper VERSION 2.2.1 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT MSVC)
  message(FATAL_ERROR "Use the Visual Studio 2022 x64 C++ toolchain on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX GR_STATIC)
add_compile_options(/W4 /WX /utf-8 /permissive- /guard:cf /EHsc)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT "/PDBALTPATH:%_PDB%")
option(GPR_ASAN "Instrument pure tests and the application with AddressSanitizer" OFF)
if(GPR_ASAN)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/DEBUG /INCREMENTAL:NO)
endif()
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
add_library(InterceptorSDK STATIC Interceptor/Interceptor.cpp)
target_include_directories(InterceptorSDK PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Interceptor")
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
add_library(RemapperCore STATIC App/Core/ControlFrame.cpp App/Core/ScriptCompiler.cpp App/Core/ScriptInstance.cpp App/Core/VisualDocument.cpp)
target_link_libraries(RemapperCore PUBLIC bcrypt)
add_library(RemapperCapture STATIC App/Capture/Remapper.cpp App/Capture/Service.cpp App/Capture/LiveTests.cpp)
target_link_libraries(RemapperCapture PUBLIC RemapperCore InterceptorSDK psapi)
file(GLOB ui_sources CONFIGURE_DEPENDS App/UI/*.cpp)
add_library(RemapperUI STATIC ${ui_sources})
target_link_libraries(RemapperUI PUBLIC RemapperCapture comctl32 comdlg32 gdi32 shell32 ole32 oleaut32 oleacc uuid dwmapi uxtheme windowscodecs)
add_executable(GamePadRemapper WIN32 App/Main.cpp App/App.rc)
target_link_libraries(GamePadRemapper PRIVATE RemapperUI)
target_link_options(GamePadRemapper PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_SOURCE_DIR}/App/App.manifest")
add_executable(RemapperCaptureTests App/Tests/CaptureTests.cpp)
target_link_libraries(RemapperCaptureTests PRIVATE InterceptorSDK)
add_executable(RemapperCoreTests App/Tests/CoreTests.cpp)
target_link_libraries(RemapperCoreTests PRIVATE RemapperCore)
add_executable(RemapperVisualTests App/Tests/VisualTests.cpp)
target_link_libraries(RemapperVisualTests PRIVATE RemapperCore)
add_executable(RemapperServiceTests App/Tests/ServiceTests.cpp App/Capture/Service.cpp)
target_link_libraries(RemapperServiceTests PRIVATE RemapperCore)
add_executable(RemapperLiveTests App/Tests/LiveTests.cpp)
target_link_libraries(RemapperLiveTests PRIVATE RemapperCapture)
add_executable(RemapperWidgetTests App/Tests/WidgetTests.cpp)
target_link_libraries(RemapperWidgetTests PRIVATE RemapperUI)
add_executable(RemapperNotificationTests App/Tests/SnapshotNotificationTests.cpp)
target_link_libraries(RemapperNotificationTests PRIVATE user32)
add_executable(RemapperParameterTests App/Tests/ParameterTests.cpp)
target_link_libraries(RemapperParameterTests PRIVATE RemapperCore)
enable_testing()
add_test(NAME SnapshotNotification COMMAND RemapperNotificationTests)
add_test(NAME Parameters COMMAND RemapperParameterTests)
add_test(NAME Capture COMMAND RemapperCaptureTests)
add_test(NAME Core COMMAND RemapperCoreTests --scripts "${CMAKE_SOURCE_DIR}/Scripts" --json "${CMAKE_BINARY_DIR}/bin/core-tests.json")
add_test(NAME Visual COMMAND RemapperVisualTests --json "${CMAKE_BINARY_DIR}/bin/visual-tests.json")
add_test(NAME Service COMMAND RemapperServiceTests)
add_test(NAME Widgets COMMAND RemapperWidgetTests --json "${CMAKE_BINARY_DIR}/bin/widget-tests.json")
